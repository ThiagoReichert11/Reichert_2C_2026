99static neopixel_color_t GetLedColor(color_mode_t mode) {
100 if (mode == MODE_RANDOM) mode = (rand() % 3) + 1;
101 switch (mode) {
102 case MODE_RED: return NeoPixelRgb2Color(255, 0, 0);
103 case MODE_YELLOW: return NeoPixelRgb2Color(255, 150, 0);
104 case MODE_GREEN: return NeoPixelRgb2Color(0, 255, 0);
105 default: return NeoPixelRgb2Color(255, 0, 0);

131static void RecordReactionISR(void) {
132 if (activo && reaction_in_progress) {
133 uint32_t elapsed_us = TimerRead(TIMER_B);
134 TimerStop(TIMER_B);
135 final_reaction_time = (int32_t)(elapsed_us / 1000);
136 reaction_in_progress = false;

195static void UartTask(void *pvParameter) {
196 uint32_t cmd;
197 while (true) {
198 if (xTaskNotifyWait(0, 0xFFFFFFFF, &cmd, portMAX_DELAY) == pdTRUE) {
199 if (cmd == 'a' || cmd == 'A') {
200 activo = !activo;
201 xTaskNotifyGive(control_task_handle);
202 } else if (cmd == '1') {
203 current_mode = MODE_RED;
204 printf("\n[MODO] Color fijo: ROJO\r\n");
205 } else if (cmd == '2') {
206 current_mode = MODE_YELLOW;
207 printf("\n[MODO] Color fijo: AMARILLO\r\n");
208 } else if (cmd == '3') {
209 current_mode = MODE_GREEN;
210 printf("\n[MODO] Color fijo: VERDE\r\n");
211 } else if (cmd == '4') {
212 current_mode = MODE_RANDOM;
213 printf("\n[MODO] Color: ALEATORIO\r\n");
214 } else if (cmd == '5') {
215 current_mode = MODE_GONOGO;
216 printf("\n[MODO] Go/No-Go (Verde: Presionar | Rojo: Ignorar)\r\n");
217 }
218 }
219 }

236static void ControlTask(void *pvParameter) {
237 bool last_activo = false;
238
239 printf("=== ENTRENADOR MULTITAREA FREERTOS ===\r\n");
240
241 while (true) {
242 if (activo) {
243 if (!last_activo) {
244 printf("\n--- DISPOSITIVO ENCENDIDO ---\r\n");
245 last_activo = true;
246 }
247
248 NeoPixelAllOff();
249 uint32_t delay_us = MIN_OFF_US + (rand() % (MAX_OFF_US - MIN_OFF_US + 1));
250 TimerStop(TIMER_A);
251 TimerUpdatePeriod(TIMER_A, delay_us);
252 TimerReset(TIMER_A);
253 TimerStart(TIMER_A);
254
255 printf("Esperando estímulo...\r\n");
256
257 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
258 TimerStop(TIMER_A);
259 if (!activo) continue;
260
261 bool is_red_gonogo = false;
262 neopixel_color_t color;
263
264 if (current_mode == MODE_GONOGO) {
265 if (rand() % 2 == 1) {
266 is_red_gonogo = true;
267 color = NeoPixelRgb2Color(255, 0, 0);
268 } else {
269 is_red_gonogo = false;
270 color = NeoPixelRgb2Color(0, 255, 0);
271 }
272 } else {
273 color = GetLedColor(current_mode);
274 }
275
276 NeoPixelAllColor(color);
277 TimerStop(TIMER_B);
278 TimerReset(TIMER_B);
279 TimerStart(TIMER_B);
280 reaction_in_progress = true;
281
282 uint32_t pressed = ulTaskNotifyTake(pdTRUE, pdMS_TO_TICKS(TIMEOUT_MS));
283 NeoPixelAllOff();
284 if (!activo) continue;
285
286 if (current_mode == MODE_GONOGO && is_red_gonogo) {
287 if (pressed) {
288 printf("¡ERROR! Falsa alarma (Presionó en ROJO)\r\n");
289 } else {
290 TimerStop(TIMER_B);
291 reaction_in_progress = false;
292 printf("¡CORRECTO! Inhibición lograda en ROJO\r\n");
293 }
294 } else {
295 if (pressed && !reaction_in_progress) {
296 printf("¡Acierto! Tiempo de reacción: %ld ms\r\n", (long)final_reaction_time);
297 } else {
298 TimerStop(TIMER_B);
299 reaction_in_progress = false;
300 printf(current_mode == MODE_GONOGO ? "Tiempo agotado. Omisión en VERDE.\r\n"
301 : "Tiempo agotado. Demasiado lento.\r\n");
302 }
303 }
304
305 if (activo) vTaskDelay(pdMS_TO_TICKS(2000));
306 } else {
307 if (last_activo) {
308 printf("\n--- DISPOSITIVO APAGADO ---\r\n");
309 last_activo = false;
310 }
311 NeoPixelAllOff();
312 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
313 }
314 }

326void app_main(void) {
327 NeoPixelInit(LED_GPIO, NUM_LEDS, led_colors);
328 NeoPixelBrightness(255);
329
330 SwitchesInit();
331 SwitchActivInt(SWITCH_1, ButtonReactionISR, NULL);
332 SwitchActivInt(SWITCH_2, Switch2Callback, NULL);
333
334 GPIOInit(EXTERNAL_BUTTON_GPIO, GPIO_INPUT);
335 GPIOInputFilter(EXTERNAL_BUTTON_GPIO);
336 GPIOActivInt(EXTERNAL_BUTTON_GPIO, ButtonReactionISR, false, NULL);
337
338 serial_config_t uart_cfg = { .port = UART_PC, .baud_rate = 115200, .func_p = UartReceiveISR, .param_p = NULL };
339 UartInit(&uart_cfg);
340
341 timer_config_t timer_a_cfg = { .timer = TIMER_A, .period = 1000000, .func_p = TimerACallback, .param_p = NULL };
342 TimerInit(&timer_a_cfg);
343
344 timer_config_t timer_b_cfg = { .timer = TIMER_B, .period = 0xFFFFFFFF, .func_p = NULL, .param_p = NULL };
345 TimerInit(&timer_b_cfg);
346
347 xTaskCreate(UartTask, "UartTask", 2048, NULL, 3, &uart_task_handle);
348 xTaskCreate(ControlTask, "ControlTask", 3072, NULL, 4, &control_task_handle);
349}
