102static void ControlLeds(uint16_t dist){
103 if (dist < 10) {
104 LedOff(LED_1);
105 LedOff(LED_2);
106 LedOff(LED_3);
107 } else if (dist >= 10 && dist < 20) {
108 LedOn(LED_1);
109 LedOff(LED_2);
110 LedOff(LED_3);
111 } else if (dist >= 20 && dist < 30) {
112 LedOn(LED_1);
113 LedOn(LED_2);

129static void TeclasTask(void *pvParameter){
130 uint8_t teclas;
131 while(true){
132 teclas = SwitchesRead();
133 if(teclas & SWITCH_1){
134 activo = !activo;
135 }
136 if(teclas & SWITCH_2){

138 }

169static void MostrarTask(void *pvParameter){
170 static uint16_t distancia_lcd = 0;
171
172 LedsInit();
173 LcdItsE0803Init();
174
175 while(true){
176 if(activo){
177 /* Los LEDs siempre responden a la distancia medida en tiempo real */
178 ControlLeds(distancia);
179
180 /* El valor del LCD solo se actualiza si hold está desactivado */
181 if(!hold){
182 distancia_lcd = distancia;
183 }
184
185 LcdItsE0803Write(distancia_lcd);
186 } else {
187 LedOff(LED_1);
188 LedOff(LED_2);
189 LedOff(LED_3);
190 LcdItsE0803Off();
191 }
192 vTaskDelay(CONFIG_PERIOD_MOSTRAR / portTICK_PERIOD_MS);
193 }
194}
