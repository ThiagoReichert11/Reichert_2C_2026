158static void MostrarTask(void *pvParameter){
159 static uint16_t distancia_lcd = 0;
160
161 LedsInit();
162 LcdItsE0803Init();
163
164 while(true){
165 if(activo){
166 /* Los LEDs siempre responden a la distancia medida en tiempo real */
167 ControlLeds(distancia);
168
169 /* El valor del LCD solo se actualiza si hold está desactivado */
170 if(!hold){
171 distancia_lcd = distancia;
172 }
173
174 LcdItsE0803Write(distancia_lcd);
175 } else {
176 LedOff(LED_1);
177 LedOff(LED_2);
178 LedOff(LED_3);
179 LcdItsE0803Off();
180 }
181 vTaskDelay(CONFIG_PERIOD_MOSTRAR / portTICK_PERIOD_MS);
182 }
