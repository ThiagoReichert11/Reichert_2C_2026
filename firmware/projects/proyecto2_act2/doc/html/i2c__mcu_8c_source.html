221bool I2C_writeBits(uint8_t devAddr, uint8_t regAddr, uint8_t bitStart, uint8_t length, uint8_t data) {
222 // 010 value to write
223 // 76543210 bit numbers
224 // xxx args: bitStart=4, length=3
225 // 00011100 mask byte
226 // 10101111 original value (sample)
227 // 10100011 original & ~mask
228 // 10101011 masked | value
229 uint8_t b = 0;
230 if (I2C_readByte(devAddr, regAddr, &b, 0) != 0) {
231 uint8_t mask = ((1 << length) - 1) << (bitStart - length + 1);
232 data <<= (bitStart - length + 1); // shift data into correct position
233 data &= mask; // zero all non-important bits in data
234 b &= ~(mask); // zero all important bits in existing byte
235 b |= data; // combine data with existing byte
236 return I2C_writeByte(devAddr, regAddr, b);
237 } else {
238 return false;
239 }
240}

268bool I2C_writeBytes(uint8_t devAddr, uint8_t regAddr, uint8_t length, uint8_t *data){
269 uint8_t *buffer = NULL;
270 i2c_master_dev_handle_t dev_handle = NULL;
271 esp_err_t ret = ESP_FAIL;
272
273 if (data == NULL || length == 0) {
274 return false;
275 }
276
277 buffer = (uint8_t *)malloc(length + 1);
278 if (buffer == NULL) {
279 return false;
280 }
281
282 buffer[0] = regAddr;
283 memcpy(&buffer[1], data, length);
284
285 ret = i2c_mcu_get_device_handle(devAddr, &dev_handle);
286 if (ret == ESP_OK) {
287 ret = i2c_master_transmit(dev_handle, buffer, length + 1, i2c_mcu_timeout_ms(0));
288 }
289
290 i2c_mcu_release_device_handle(dev_handle);
291 free(buffer);
292 return (ret == ESP_OK);
293}
