108void displayWrite(uint32_t data, uint8_t digits, gpioConf_t *bcd_pins, gpioConf_t *lcd_digit_pins)
109{
110 bool is_float = (data != (uint32_t)data);
111 bool is_out_of_range = (data >= 1000 || digits > 3);
112
113 // Manejo de condición de error
114 if (is_float || is_out_of_range)
115 {
116 writeBcdToGpio(0, bcd_pins);
117
118 for (uint8_t i = 0; i < digits; i++)
119 {
120 GPIOToggle(lcd_digit_pins[i].pin);
121 }
122
123 return;
124 }
125
126 uint8_t bcd_array[digits];
127
128 // Descomposición del número a BCD
129 convertToBcdArray(data, digits, bcd_array);
130
131 // Apagado de los dígitos del display para evitar solapamientos
132 for (uint8_t j = 0; j < 3; j++)
133 {
134 GPIOOff(lcd_digit_pins[j].pin);
135 }
136
137 // Multiplexado del dato por pantalla
138 for (uint8_t i = 0; i < digits; i++)
139 {
140 writeBcdToGpio(bcd_array[i], bcd_pins);
141
142 GPIOOn(lcd_digit_pins[i].pin);
143
144 GPIOOff(lcd_digit_pins[i].pin);
145 }
146}

156void app_main(void)
157{
158 // Dato a mostrar (3 dígitos)
159 uint32_t data = 2;
160 uint8_t digits = 3;
161
163 gpioConf_t bcd_pins[4] = {
164 {GPIO_20, GPIO_OUTPUT}, // b0
165 {GPIO_21, GPIO_OUTPUT}, // b1
166 {GPIO_22, GPIO_OUTPUT}, // b2
167 {GPIO_23, GPIO_OUTPUT} // b3
168 };
169
171 gpioConf_t lcd_digits[3] = {
172 {GPIO_19, GPIO_OUTPUT}, // Dígito 1
173 {GPIO_18, GPIO_OUTPUT}, // Dígito 2
174 {GPIO_9, GPIO_OUTPUT} // Dígito 3
175 };
176
177 // Inicialización del hardware BCD
178 for (uint8_t i = 0; i < 4; i++) {
179 GPIOInit(bcd_pins[i].pin, bcd_pins[i].dir);
180 }
181
182 // Inicialización del hardware de control de dígitos
183 for (uint8_t i = 0; i < digits; i++) {
184 GPIOInit(lcd_digits[i].pin, lcd_digits[i].dir);
185 }
186
187 // Mostrar el número en la pantalla
188 displayWrite(data, digits, bcd_pins, lcd_digits);
189}
