117static void ControlLeds(uint16_t dist){
118 if (dist < 10) {
119 LedOff(LED_1);
120 LedOff(LED_2);
121 LedOff(LED_3);
122 } else if (dist >= 10 && dist < 20) {
123 LedOn(LED_1);
124 LedOff(LED_2);
125 LedOff(LED_3);
126 } else if (dist >= 20 && dist < 30) {
127 LedOn(LED_1);
128 LedOn(LED_2);
129 LedOff(LED_3);
130 } else {
131 LedOn(LED_1);

169void FuncUart(void *param){
170 uint8_t tecla;
171 UartReadByte(UART_PC, &tecla);
172 if(tecla == 'O' || tecla == 'o'){
173 activo = !activo;
174 }
175 if(tecla == 'H' || tecla == 'h'){
176 hold = !hold;
177 }
178 if(tecla == 'I' || tecla == 'i'){
179 pulgadas = !pulgadas;
180 distancia_maxima = 0; /* Reinicia el máximo al cambiar la escala/unidad */
181 }
182 if(tecla == 'M' || tecla == 'm'){
183 modo_max = !modo_max;
184 }
185}

206static void MedirTask(void *pvParameter){
207 uint16_t valor_a_enviar;
208
209 while(true){
210 if(activo){
211 /* Medición en tiempo real según la unidad activa */
212 if(pulgadas){
213 distancia = HcSr04ReadDistanceInInches();
214 } else {
215 distancia = HcSr04ReadDistanceInCentimeters();
216 }
217
218 /* Actualización de la distancia máxima registrada */
219 if(distancia > distancia_maxima){
220 distancia_maxima = distancia;
221 }
222
223 /* Si no está en HOLD, actualiza el valor de retención */
224 if(!hold){
225 distancia_retenida = distancia;
226 }
227
228 /* Selección del valor a enviar con condición tradicional */
229 if(modo_max){
230 valor_a_enviar = distancia_maxima;
231 } else {
232 valor_a_enviar = distancia_retenida;
233 }
234
235 /* Transmisión por UART */
236 UartSendString(UART_PC, (char *)UartItoa(valor_a_enviar, 10));
237
238 if(pulgadas){
239 UartSendString(UART_PC, " in\r\n");
240 } else {
241 UartSendString(UART_PC, " cm\r\n");
242 }
243 }
244 vTaskDelay(CONFIG_PERIOD_MEDIR / portTICK_PERIOD_MS);
245 }
246}

256static void MostrarTask(void *pvParameter){
257 uint16_t valor_a_mostrar;
258
259 LedsInit();
260 LcdItsE0803Init();
261
262 while(true){
263 if(activo){
264 /* Los LEDs responden a la medición real instantánea */
265 ControlLeds(distancia);
266
267 /* Selección del valor a mostrar con condición tradicional */
268 if(modo_max){
269 valor_a_mostrar = distancia_maxima;
270 } else {
271 valor_a_mostrar = distancia_retenida;
272 }
273
274 LcdItsE0803Write(valor_a_mostrar);
275 } else {
276 LedOff(LED_1);
277 LedOff(LED_2);
278 LedOff(LED_3);
279 LcdItsE0803Off();
280 }
281 vTaskDelay(CONFIG_PERIOD_MOSTRAR / portTICK_PERIOD_MS);
282 }
